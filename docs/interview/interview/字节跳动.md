2. 二叉树所有根到叶子路径组成的数字之和

```js
function sumTreePath(node) {
    if (!node) return 0
    
    let arr = []
    function loop(node, n) {
        if (!node.left && !node.right) {
            arr.push(n+=(''+node.val))
        }
        
        if (node.left) {
           loop(node.left, node.val)
        }
        if (node.right) {
            loop(node.right, node.val)
        }
        return arr
    }
    loop(node)
    let sum = arr.reduce((item1, item2) => Number(item1)+Number(item2))
    return sum
}
```
